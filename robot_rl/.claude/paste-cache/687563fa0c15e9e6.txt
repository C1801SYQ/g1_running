    def _validate_scale_range(params, name, *, allow_negative=False, allow_zero=True):
        if params is None:
            return
        if not isinstance(params, tuple) or len(params) != 2:
            raise TypeError(f"{name} must be a tuple of two numbers, got: {params}")
        low, high = params
        if low > high:
            raise ValueError(f"{name} lower bound must be <= upper bound, got: {params}")
        if not allow_negative and low < 0:
            raise ValueError(f"{name} lower bound must be >= 0, got: {params}")
        if not allow_zero and low <= 0:
            raise ValueError(f"{name} lower bound must be > 0, got: {params}")

补丁 2：
